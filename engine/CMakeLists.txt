cmake_minimum_required(VERSION 3.16)
project(fxbox_engine VERSION 1.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Pi 4 (Cortex-A72) tuning when building natively on the device.
option(FXBOX_TUNE_PI4 "Tune for Raspberry Pi 4 (Cortex-A72)" OFF)
set(FXBOX_FLAGS -O2 -Wall -Wextra -Wno-unused-parameter -fno-math-errno)
if(FXBOX_TUNE_PI4)
  list(APPEND FXBOX_FLAGS -mcpu=cortex-a72)
endif()

add_library(fxbox_core STATIC src/engine_core.cpp src/tempo.cpp)
target_include_directories(fxbox_core PUBLIC src)
target_compile_options(fxbox_core PRIVATE ${FXBOX_FLAGS} -Wno-unused-variable -Wno-unused-but-set-variable)
target_compile_definitions(fxbox_core PUBLIC FXBOX_VERSION="${PROJECT_VERSION}")

option(FXBOX_BUILD_ENGINE "Build the JACK engine executable" ON)
if(FXBOX_BUILD_ENGINE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JACK REQUIRED IMPORTED_TARGET jack)
  pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
  find_package(Threads REQUIRED)
  add_executable(fxbox-engine src/main.cpp src/midi_input.cpp)
  target_compile_options(fxbox-engine PRIVATE ${FXBOX_FLAGS})
  target_link_libraries(fxbox-engine PRIVATE fxbox_core PkgConfig::JACK PkgConfig::ALSA Threads::Threads)
  install(TARGETS fxbox-engine RUNTIME DESTINATION bin)
endif()

option(FXBOX_BUILD_TESTS "Build offline tests" ON)
if(FXBOX_BUILD_TESTS)
  add_executable(test_dsp tests/test_dsp.cpp)
  target_link_libraries(test_dsp PRIVATE fxbox_core)
  target_compile_options(test_dsp PRIVATE ${FXBOX_FLAGS})
  add_executable(test_tempo tests/test_tempo.cpp)
  target_link_libraries(test_tempo PRIVATE fxbox_core)
  target_compile_options(test_tempo PRIVATE ${FXBOX_FLAGS})
  enable_testing()
  add_test(NAME dsp COMMAND test_dsp)
  add_test(NAME tempo COMMAND test_tempo)
endif()
